    // admin.html: só ADM lê/aprova/apaga. Aprovada vai para mc_vagas (leitura pública; o app mostra até "expira").
    function mcTxt(d, k, mn, mx) {
      return d[k] is string && d[k].size() >= mn && d[k].size() <= mx;
    }
    match /mc_vagas_pendentes/{id} {
      allow create: if request.resource.data.keys().hasOnly(['empresa','uf','cidade','bairro','cargo','descricao','tipo','comoTipo','como','contato','criadoEm'])
        && request.resource.data.keys().hasAll(['empresa','uf','cidade','cargo','tipo','comoTipo','como','contato','criadoEm'])
        && mcTxt(request.resource.data, 'empresa', 2, 80)
        && request.resource.data.uf is string && request.resource.data.uf.matches('^[A-Z]{2}$')
        && mcTxt(request.resource.data, 'cidade', 2, 60)
        && (!('bairro' in request.resource.data) || mcTxt(request.resource.data, 'bairro', 0, 60))
        && mcTxt(request.resource.data, 'cargo', 3, 80)

        && request.resource.data.comoTipo in ['whatsapp','email','endereco']
