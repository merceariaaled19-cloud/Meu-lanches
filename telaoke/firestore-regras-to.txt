    // ===== TelaOkê Publicidades (telaoke/) — modelo operador único (v3, 2026-11-24) =====
    // mlAdmin() = as 3 contas de administrador (já definida no topo das regras).
    // TVs não fazem login: leem o próprio to_telas/{código}, os anúncios ligados e to_config, e só atualizam "visto".
    match /to_config/{d} {
      allow read: if true;
      allow write: if mlAdmin();
    }
    match /to_lojas/{id} {
      allow read, write: if mlAdmin();
    }
    match /to_anuncios/{id} {
      allow read: if resource.data.ativo == true || mlAdmin();
      allow create, update: if mlAdmin() && request.resource.data.get('img', '').size() < 450000;
      allow delete: if mlAdmin();
    }
    match /to_telas/{codigo} {
      allow get: if true;
      allow list: if mlAdmin();
      allow create: if mlAdmin() && codigo.size() == 6;
      allow update: if mlAdmin()
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['visto']) && request.resource.data.visto == request.time);
      allow delete: if mlAdmin();
    }
    match /to_interessados/{id} {
      allow create: if request.resource.data.keys().hasOnly(['loja','zap','origem','criadoEm'])
        && request.resource.data.loja is string && request.resource.data.loja.size() >= 2 && request.resource.data.loja.size() <= 60
        && request.resource.data.zap is string && request.resource.data.zap.matches('^55[0-9]{10,11}$')
        && request.resource.data.get('origem', 'site') is string && request.resource.data.get('origem', 'site').size() <= 20
        && request.resource.data.criadoEm == request.time;
      allow read, update, delete: if mlAdmin();
    }
    // coleções do modelo antigo (revenda): só o ADM
    match /to_clientes/{id} {
      allow read, write: if mlAdmin();
    }
    match /to_validade/{id} {
      allow read: if true;
      allow write: if mlAdmin();
    }
    match /to_pagamentos/{p} {
      allow read, write: if mlAdmin();
    }
